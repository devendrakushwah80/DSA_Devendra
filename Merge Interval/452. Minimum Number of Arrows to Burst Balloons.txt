class Solution:
    def findMinArrowShots(self, points: List[List[int]]) -> int:
        if not points:
            return 0
        points.sort()
        count = 1
        start = points[0][0]
        end = points[0][1]
        for i in range(1, len(points)):
            s, e = points[i]
            if s <= end:
                start = max(start, s)
                end = min(end, e)
            else:
                count += 1
                start = s
                end = e
        return count
