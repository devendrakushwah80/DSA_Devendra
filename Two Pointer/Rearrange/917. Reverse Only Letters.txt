class Solution:
    def reverseOnlyLetters(self, s: str) -> str:
        s = list(s)
        l = 0
        h = len(s)-1
        while l<h:
            while l<h and not s[l].isalpha():
                l+=1
            while l<h and not s[h].isalpha():
                h-=1
            s[l],s[h]=s[h],s[l]
            l+=1
            h-=1
        return "".join(s)
